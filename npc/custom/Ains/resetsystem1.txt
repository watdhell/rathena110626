
-	script	RebirthSystem1	-1,{

OnInit:
	// ==========================================
	// REBIRTH CONFIGURATION
	// ==========================================

	.RequiredBase = 99;
	.RequiredJob = 50;

	.ResetItem = 7227;
	.ResetAmount = 1;

	.BaseStatReward = 100;
	.MaxRebirth = 0; // 0 = Unlimited

	// GM configuration
	.GMLevel = 80;

	// Milestone configuration
	setarray .Milestone[0],10,20,30,50,100;

	setarray .MilestoneStat[0],
		100,200,300,500,1000;

	setarray .MilestoneItem[0],
		607,607,607,607,607;

	setarray .MilestoneAmount[0],
		10,20,30,50,100;

	// Permanent bonus package item.
	setarray .PermanentItem[0],
		501,501,501,501,501;

	setarray .PermanentAmount[0],
		1,2,3,5,10;

	// Ranking
	.RankLimit = 10;

	end;

OnRebirth:
	mes "[ Rebirth Master ]";
	mes "Your current rebirth count:";
	mes "^0000FF" + .@rebirth + "^000000";
	mes "Required Base Level: " + .RequiredBase;
	mes "Required Job Level: " + .RequiredJob;
	mes " ";
	mes "Required item:";
	mes getitemname(.ResetItem) + " x " + .ResetAmount;
	next;

	if (select("Rebirth now:Cancel") == 2)
		close;

	// Character identity
	.@char_id = getcharid(0);
	.@account_id = getcharid(3);
	.@name$ = strcharinfo(0);

	// Prevent invalid character states
	if (Hp <= 0) {
		mes "[ Rebirth Master ]";
		mes "You cannot rebirth while dead.";
		close;
	}

	if (checkvending() || checkchatting()) {
		mes "Close your shop or chat room first.";
		close;
	}

	if (BaseLevel < .RequiredBase ||
	    JobLevel < .RequiredJob) {
		mes "You have not reached the required levels.";
		close;
	}

	if (countitem(.ResetItem) < .ResetAmount) {
		mes "You do not have the required item.";
		close;
	}

	// Read authoritative SQL count
	query_sql(
		"SELECT rebirth_count FROM rebirth_status " +
		"WHERE char_id = " + .@char_id,
		.@rebirth
	);

	if (.@rebirth < 0)
		.@rebirth = 0;

	if (.MaxRebirth > 0 && .@rebirth >= .MaxRebirth) {
		mes "You have reached the maximum rebirth count.";
		close;
	}

	.@next = .@rebirth + 1;

	// Generate a transaction token
	.@token = gettimetick(2) * 100000 + .@char_id;

	// Acquire per-character SQL lock.
	query_sql(
		"INSERT IGNORE INTO rebirth_locks " +
		"(char_id,account_id,transaction_id) VALUES (" +
		.@char_id + "," + .@account_id + "," + .@token + ")"
	);

	.@lock = 0;

	query_sql(
		"SELECT transaction_id FROM rebirth_locks " +
		"WHERE char_id = " + .@char_id,
		.@lock
	);

	if (.@lock != .@token) {
		mes "Another rebirth operation is active.";
		close;
	}

	// Prepare SQL journal.
	query_sql(
		"INSERT INTO rebirth_journal " +
		"(char_id,account_id,reset_number,state,item_id,item_amount) " +
		"VALUES (" +
		.@char_id + "," +
		.@account_id + "," +
		.@next + ",'PREPARED'," +
		.ResetItem + "," +
		.ResetAmount + ")"
	);

	.@tx = 0;

	query_sql("SELECT LAST_INSERT_ID()", .@tx);

	if (.@tx <= 0) {
		query_sql(
			"DELETE FROM rebirth_locks WHERE char_id = " +
			.@char_id + " AND transaction_id = " + .@token
		);
		mes "Could not prepare the transaction.";
		close;
	}

	// Recheck the item immediately before consumption.
	if (countitem(.ResetItem) < .ResetAmount) {
		query_sql(
			"UPDATE rebirth_journal SET state='FAILED' " +
			"WHERE transaction_id=" + .@tx
		);
		query_sql(
			"DELETE FROM rebirth_locks WHERE char_id=" +
			.@char_id + " AND transaction_id=" + .@token
		);
		mes "Required item is missing.";
		close;
	}

	// Consume item.
	delitem .ResetItem, .ResetAmount;

	query_sql(
		"UPDATE rebirth_journal SET state='ITEM_CONSUMED' " +
		"WHERE transaction_id=" + .@tx
	);

	// Determine normal and milestone rewards.
	.@stat_reward = .BaseStatReward;
	.@reward_id = 0;
	.@reward_amount = 0;
	.@bonus_points = 0;
	.@permanent_id = 0;
	.@permanent_amount = 0;

	for (.@i = 0; .@i < getarraysize(.Milestone); .@i++) {
		if (.@next == .Milestone[.@i]) {
			.@stat_reward += .MilestoneStat[.@i];
			.@reward_id = .MilestoneItem[.@i];
			.@reward_amount = .MilestoneAmount[.@i];
			.@bonus_points = .MilestoneStat[.@i];
			.@permanent_id = .PermanentItem[.@i];
			.@permanent_amount = .PermanentAmount[.@i];
			break;
		}
	}

	// Record planned rewards.
	query_sql(
		"UPDATE rebirth_journal SET " +
		"stat_reward=" + .@stat_reward + "," +
		"reward_item_id=" + .@reward_id + "," +
		"reward_amount=" + .@reward_amount +
		" WHERE transaction_id=" + .@tx
	);

	// Reset Base/Job Level, EXP and skill points.
	// Preserves learned skills and existing stats.
	resetlvl 2;

	// Add configurable status points.
	StatusPoint += .@stat_reward;

	query_sql(
		"UPDATE rebirth_journal SET state='LEVEL_RESET' " +
		"WHERE transaction_id=" + .@tx
	);

	// Save authoritative rebirth count.
	query_sql(
		"INSERT INTO rebirth_status " +
		"(char_id,account_id,rebirth_count,bonus_status_points) " +
		"VALUES (" +
		.@char_id + "," +
		.@account_id + "," +
		.@next + "," +
		.@bonus_points + ") " +
		"ON DUPLICATE KEY UPDATE " +
		"account_id=VALUES(account_id)," +
		"rebirth_count=VALUES(rebirth_count)," +
		"bonus_status_points=bonus_status_points+" +
		.@bonus_points
	);

	// Store reset history.
	query_sql(
		"INSERT INTO rebirth_logs " +
		"(char_id,account_id,char_name,reset_number," +
		"base_level_before,job_level_before," +
		"consumed_item_id,consumed_amount," +
		"stat_points_reward,reward_item_id,reward_item_amount) " +
		"VALUES (" +
		.@char_id + "," +
		.@account_id + ",'" +
		escape_sql(.@name$) + "'," +
		.@next + "," +
		.RequiredBase + "," +
		.RequiredJob + "," +
		.ResetItem + "," +
		.ResetAmount + "," +
		.@stat_reward + "," +
		.@reward_id + "," +
		.@reward_amount + ")"
	);

	// Mark committed before delivering rewards.
	query_sql(
		"UPDATE rebirth_journal SET state='COMMITTED' " +
		"WHERE transaction_id=" + .@tx
	);

	// Give ordinary rewards.
	if (.@reward_id > 0 && .@reward_amount > 0)
		getitem .@reward_id, .@reward_amount;

	// Give permanent milestone package.
	if (.@permanent_id > 0 && .@permanent_amount > 0)
		getitem .@permanent_id, .@permanent_amount;

	// Release lock.
	query_sql(
		"DELETE FROM rebirth_locks WHERE char_id=" +
		.@char_id + " AND transaction_id=" + .@token
	);

	dispbottom "Rebirth completed!";
	dispbottom "Your new rebirth count: " + .@next;
	dispbottom "Status points received: " + .@stat_reward;

	close;
}


aegis,86,72,4	script	Rebirth God	4_F_TELEPORTER,{

	mes "[ Rebirth Master ]";

	.@cid = getcharid(0);

	query_sql(
		"SELECT rebirth_count FROM rebirth_status " +
		"WHERE char_id = " + .@cid,
		.@count
	);

	mes "Welcome, adventurer!";
	mes " ";
	mes "Your Rebirth Count: ^0000FF" + .@count + "^000000";

	next;

	switch(select(
		"Rebirth Character",
		"Rebirth Ranking",
		"View Rebirth Rewards",
		"Cancel"
	)) {
		case 1:
			donpcevent "RebirthSystem1::OnRebirth";
			end;

		case 2:
			donpcevent "RebirthRanking1::OnShow";
			end;

		case 3:
			mes "[ Rebirth Rewards ]";
			mes "10 / 20 / 30 / 50 / 100";
			mes "Milestone rewards are granted once.";
			close;

		default:
			close;
	}
}
